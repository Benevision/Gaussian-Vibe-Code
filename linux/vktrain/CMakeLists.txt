# SplatWork Vulkan trainer (stage 1: kernels + numerical checks). See docs/dev/GPU_SUPPORT.md,
# "Decision (2026-11-13): a portable Vulkan trainer".
#
#   cmake -S . -B build -G Ninja && cmake --build build && ./build/vktrain-selftest
#
# GLSL compute shaders (shaders/*.comp) are compiled to SPIR-V with glslangValidator at build time and embedded
# as C arrays; kernels that add floats atomically are built twice (compare-and-swap loop, and native
# GL_EXT_shader_atomic_float), the variant is chosen at run time from the device's features.
cmake_minimum_required(VERSION 3.18)
project(splatwork_vktrain LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Vulkan REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)

set(KERNELS preprocess preprocess_bwd radix_hist radix_scatter scan_block scan_add gather_counts duplicate_keys
            tile_ranges raster_fwd raster_bwd ssim_prep blur ssim_terms loss_grad adam atomic_probe)
set(ATOMIC_KERNELS raster_bwd ssim_terms atomic_probe)

set(SPV_DIR ${CMAKE_CURRENT_BINARY_DIR}/spv)
file(MAKE_DIRECTORY ${SPV_DIR})
set(SPV_HEADERS "")
set(REG_INCLUDES "")
set(REG_ENTRIES "")
set(NATIVE_ENTRIES "")
file(GLOB SHADER_INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.glsl)
foreach(k ${KERNELS})
  set(variants ${k})
  if(k IN_LIST ATOMIC_KERNELS)
    list(APPEND variants ${k}_native)
    string(APPEND NATIVE_ENTRIES "\"${k}\", ")
  endif()
  foreach(v ${variants})
    set(defs "")
    if(v MATCHES "_native$")
      set(defs -DNATIVE_FLOAT_ATOMICS=1)
    endif()
    add_custom_command(
      OUTPUT ${SPV_DIR}/${v}.h
      COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.2 ${defs} -I${CMAKE_CURRENT_SOURCE_DIR}/shaders
              --vn spv_${v} -o ${SPV_DIR}/${v}.h ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${k}.comp
      DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${k}.comp ${SHADER_INCLUDES}
      COMMENT "GLSL -> SPIR-V ${v}"
      VERBATIM)
    list(APPEND SPV_HEADERS ${SPV_DIR}/${v}.h)
    string(APPEND REG_INCLUDES "#include \"${v}.h\"\n")
    string(APPEND REG_ENTRIES "    {\"${v}\", spv_${v}, sizeof(spv_${v})},\n")
  endforeach()
endforeach()
file(CONFIGURE OUTPUT ${SPV_DIR}/shaders_registry.cpp @ONLY CONTENT [=[
// Generated by CMakeLists.txt: the embedded SPIR-V kernels.
#include <cstdint>
#include <cstring>
#include "shaders.h"
@REG_INCLUDES@
namespace vkt {
static const ShaderBlob kBlobs[] = {
@REG_ENTRIES@};
static const char* const kNative[] = {@NATIVE_ENTRIES@ nullptr};
const ShaderBlob* findShader(const char* name) {
    for (const auto& b : kBlobs)
        if (std::strcmp(b.name, name) == 0) return &b;
    return nullptr;
}
bool hasNativeVariant(const char* name) {
    for (const char* const* k = kNative; *k; k++)
        if (std::strcmp(*k, name) == 0) return true;
    return false;
}
}  // namespace vkt
]=])

add_library(vktrain_core STATIC
  src/vk.cpp src/gpu_splat.cpp src/cpu_ref.cpp ${SPV_DIR}/shaders_registry.cpp ${SPV_HEADERS})
target_include_directories(vktrain_core PUBLIC src PRIVATE ${SPV_DIR})
target_link_libraries(vktrain_core PUBLIC Vulkan::Vulkan)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(vktrain_core PRIVATE -Wall -Wextra)
endif()

add_executable(splatwork-vk-train src/main.cpp)
target_link_libraries(splatwork-vk-train PRIVATE vktrain_core)

add_executable(vktrain-selftest src/selftest.cpp)
target_link_libraries(vktrain-selftest PRIVATE vktrain_core)

enable_testing()
add_test(NAME vktrain-selftest COMMAND vktrain-selftest)
